# M1b 見た目の実装 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. このマイルストーンは表示層で「ユーザーの画面で美しいか」を問うため、サブエージェント駆動ではなくインライン実行(各所でユーザー官能チェック)を採用する。

**Goal:** M1aで完成した決定論シミュレーション層の状態を、Sunny Landのフリー素材(キツネ33x32・カエル35x32)で本物のスプライトアニメとして描画し、コート背景・スコアUI・CRTフィルター・表示オプションを備えた「ゲームらしい画面」を作る。

**Tech Stack:** Godot 4.6-stable、GDScript、AnimatedSprite2D、SpriteFrames(コード構築)、CanvasLayer+gdshader(CRT)、既存project.godotの整数倍viewportストレッチ。

## Global Constraints

- 内部解像度は640x360(spec 4.1、仮確定・M1で官能確認)。project.godotに設定済み(viewport 640x360、stretch mode=viewport/aspect=keep/scale_mode=integer、texture_filter=0=nearest)。
- 表示は整数倍拡大(1080p=3倍/1440p=4倍/4K=6倍、spec 4.1)。CRTは整数倍拡大後の出力に適用(spec 4.3)。
- 合格基準は「遊べる」ではなく「ユーザーの画面で美しい」。可愛さ・気持ちよさの判定権は100%ユーザー(spec 品質の掟)。
- プレースホルダー(四角・丸)でゲーム画面を作ることは禁止。ただしSIM DEBUG VIEWは開発計器として例外(既存合意)。
- sim層(src/sim/)はfloat禁止・int専守で不可侵。表示層(src/display/)はfloat使用可。
- トラックA素材は非公開の開発用。本番はM4でトラックBに差し替え。素材出典はassets/third_party/README.md記載のCC0。
- class_name不使用・全てpreload参照(ヘッドレスのグローバルクラスキャッシュ非依存、既存の掟)。

```gdscript
# 個別フレームPNGからSpriteFramesをコード構築する。テクスチャ依存のため
# ヘッドレス自動テストの対象外(実描画はゲーム起動時のユーザー官能チェックで検証)。
extends RefCounted

const FOX := "res://assets/third_party/sunny_land/PNG/sprites/player"
const FROG := "res://assets/third_party/sunny_land/PNG/sprites/frog"

static func _add(sf: SpriteFrames, anim: String, tmpl: String, count: int, fps: float, loop: bool) -> void:
	sf.add_animation(anim)
	sf.set_animation_speed(anim, fps)
	sf.set_animation_loop(anim, loop)
	for i in range(1, count + 1):
		var tex: Texture2D = load(tmpl % i)
		sf.add_frame(anim, tex)

static func build_fox() -> SpriteFrames:
	var sf := SpriteFrames.new()
	sf.remove_animation("default")
	_add(sf, "idle", FOX + "/idle/player-idle-%d.png", 4, 8.0, true)
	_add(sf, "run", FOX + "/run/player-run-%d.png", 6, 12.0, true)
	_add(sf, "jump", FOX + "/jump/player-jump-%d.png", 2, 6.0, false)
	_add(sf, "crouch", FOX + "/crouch/player-crouch-%d.png", 2, 8.0, false)
	_add(sf, "hurt", FOX + "/hurt/player-hurt-%d.png", 2, 8.0, false)
	return sf

static func build_frog() -> SpriteFrames:
	var sf := SpriteFrames.new()
	sf.remove_animation("default")
	_add(sf, "idle", FROG + "/idle/frog-idle-%d.png", 4, 8.0, true)
	_add(sf, "jump", FROG + "/jump/frog-jump-%d.png", 2, 6.0, false)
	# カエルにrun素材は無いのでidleを流用(仮素材段階)
	_add(sf, "run", FROG + "/idle/frog-idle-%d.png", 4, 10.0, true)
	_add(sf, "crouch", FROG + "/idle/frog-idle-%d.png", 4, 8.0, true)
	_add(sf, "hurt", FROG + "/idle/frog-idle-%d.png", 4, 8.0, true)
	return sf
```

```gdscript
# 本番ゲームビュー。sim状態を読んでスプライトを駆動する。表示層(float可)。
extends Node2D

const SimConfig := preload("res://src/sim/sim_config.gd")
const SimState := preload("res://src/sim/sim_state.gd")
const Simulation := preload("res://src/sim/simulation.gd")
const ViewTransform := preload("res://src/display/view_transform.gd")
const SpriteFactory := preload("res://src/display/sprite_factory.gd")
const AnimSelect := preload("res://src/display/anim_select.gd")

var cfg
var state
var _sprites: Array = []  # AnimatedSprite2D×4
var _ball: Sprite2D

func _ready() -> void:
	cfg = SimConfig.new()
	if not cfg.valid:
		push_error("rules.json読み込み失敗")
		return
	state = SimState.new()
	Simulation.reset_rally(state, cfg, 0)
	Engine.physics_ticks_per_second = cfg.tick_rate
	var fox := SpriteFactory.build_fox()
	var frog := SpriteFactory.build_frog()
	for i in 4:
		var s := AnimatedSprite2D.new()
		# チーム0(左, index0,1)=キツネ、チーム1(右)=カエル(仮の色分け・種分け)
		s.sprite_frames = fox if Simulation.team_of(i) == 0 else frog
		s.centered = true
		s.offset = Vector2(0, -16)  # 足元(下端)をノード原点に合わせる(32px高の半分)
		s.play("idle")
		$Players.add_child(s)
		_sprites.append(s)
	_ball = $Ball
	_ball.centered = true

func _physics_process(_delta: float) -> void:
	var input := 0
	if Input.is_key_pressed(KEY_LEFT): input |= Simulation.IN_LEFT
	if Input.is_key_pressed(KEY_RIGHT): input |= Simulation.IN_RIGHT
	if Input.is_key_pressed(KEY_Z) or Input.is_key_pressed(KEY_SPACE): input |= Simulation.IN_JUMP
	if Input.is_key_pressed(KEY_X): input |= Simulation.IN_ACTION
	if Input.is_key_pressed(KEY_C): input |= Simulation.IN_SWITCH
	Simulation.tick(state, [input, 0], cfg)
	_sync_sprites()

func _sync_sprites() -> void:
	for i in _sprites.size():
		var p = state.players[i]
		var spr: AnimatedSprite2D = _sprites[i]
		spr.position = ViewTransform.pos_of(p)
		# Task2でアニメ選択・反転を接続する。Task1ではidle固定
	_ball.position = Vector2(ViewTransform.to_px(state.ball_x), ViewTransform.to_px(state.ball_y))
```

注記: Task1時点ではanim_select.gd未実装のため`_sync_sprites`はidle固定・位置のみ反映。`const AnimSelect`の行はTask2まではコメントアウトしておく(未定義参照回避)。Ballのtextureはgame_view.tscnでcherry.png(24x24相当の丸い実物)を指定し、□を避ける。

`src/display/anim_select.gd`:
```gdscript
# プレイヤーのsim状態(int)からアニメ名・向きを選ぶ純粋関数。
# 状態を読むだけ(副作用なし)。表示層だがテクスチャ非依存なのでヘッドレステスト可能。
extends RefCounted

# 優先順位: 空中=jump > 接地ヒット硬直=crouch > 移動=run > 静止=idle
static func anim_for(p) -> String:
	if p.on_ground == 0:
		return "jump"
	if p.hit_cooldown > 0:
		return "crouch"
	if p.vx != 0:
		return "run"
	return "idle"

# チーム0(左)は右向き=反転なし、チーム1(右)は左向き=反転
static func flip_for_team(team: int) -> bool:
	return team == 1
```

Sunny Landの背景素材(back.png=空384x240)を空色の遠景に、地面はコート下部を落ち着いた色帯+tilesetの草ラインで表現。ネットは中央の縦線+上端の白帯。仮素材段階なので「ゲームらしい」最小限。□回避のため単色ベタ塗りの巨大矩形は使わず、素材テクスチャ(back.png)をタイル/拡大配置する。

`src/display/court.gd`:
```gdscript
# コート背景・地面・ネットの描画。表示層(float可)。sim_configの寸法(fp)を読む。
extends Node2D

const ViewTransform := preload("res://src/display/view_transform.gd")
const BACK := "res://assets/third_party/sunny_land/PNG/environment/layers/back.png"

var cfg
var _back: Texture2D

func setup(config) -> void:
	cfg = config
	_back = load(BACK)
	queue_redraw()

func _draw() -> void:
	if cfg == null:
		return
	var w := ViewTransform.to_px(cfg.court_width)
	var fy := ViewTransform.to_px(cfg.floor_y)
	# 遠景(空): back.pngを画面幅に合わせて配置(640x360、backは384x240なので拡大)
	if _back != null:
		draw_texture_rect(_back, Rect2(0, 0, w, 360), false)
	# 地面帯(草地の色。単色矩形だが背景テクスチャの上の地面として機能)
	draw_rect(Rect2(0, fy, w, 360.0 - fy), Color(0.36, 0.28, 0.20))
	draw_line(Vector2(0, fy), Vector2(w, fy), Color(0.45, 0.62, 0.30), 2.0)
	# ネット
	var nx := ViewTransform.to_px(cfg.net_x)
	var nty := ViewTransform.to_px(cfg.net_top_y)

# 画面解像度からアスペクト維持の最大整数倍率(最低1)
static func max_integer_scale(screen_w: int, screen_h: int, base_w: int, base_h: int) -> int:
	var sx: int = screen_w / base_w
	var sy: int = screen_h / base_h
	return max(1, min(sx, sy))

static func clamp_intensity(v: float) -> float:
	return clampf(v, 0.0, 1.0)
```

- [ ] **Step 5: crt.gdshaderを実装**

整数倍拡大後の最終画面に対して走る全画面シェーダー。スキャンラインは実画面の物理行を基準にし、強度uniformで暗線の濃さを制御。強度0でほぼ素の絵。

`src/display/crt.gdshader`:
```glsl
shader_type canvas_item;

// スキャンライン強度(0=なし, 1=最大)。ドット1物理行ごとに暗線。
uniform float intensity : hint_range(0.0, 1.0) = 0.0;
// 走査線の周期(物理px)。1で1行ごと。
uniform float line_period = 1.0;

void fragment() {
	vec4 col = texture(TEXTURE, UV);
	// FRAGCOORD.y は物理ピクセル座標。整数倍拡大後の実画面行に一致する。
	float phase = mod(FRAGCOORD.y, line_period * 2.0);
	float dark = step(line_period, phase); // 1行おきに暗
	float factor = 1.0 - intensity * dark * 0.5;
	COLOR = vec4(col.rgb * factor, col.a);
}
```

注記: このシェーダーはColorRect(全画面)にShaderMaterialとして貼り、`TEXTURE`はゲーム描画のスクリーン結果に相当させる。stretch mode=viewportのため、CRT用ColorRectはゲームより手前の独立CanvasLayerに置き、`BackBufferCopy`+`SCREEN_TEXTURE`方式で最終出力を取り込む実装に置き換える(実装時にBackBufferCopyノードを併用)。強度uniformの受け渡しはcrt_layer.gdが行う。

# 表示設定の適用と永続化。表示層。
extends RefCounted

const DisplayScale := preload("res://src/display/display_scale.gd")
const BASE_W := 640
const BASE_H := 360
const PATH := "user://settings.cfg"

static func apply_window(win: Window, scale: int, fullscreen: bool) -> void:
	if fullscreen:
		win.mode = Window.MODE_FULLSCREEN
	else:
		win.mode = Window.MODE_WINDOWED
		win.size = Vector2i(BASE_W * scale, BASE_H * scale)

static func default_dict() -> Dictionary:
	return {"window_scale": 2, "fullscreen": false, "crt_on": false, "crt_intensity": 0.0}

static func save(d: Dictionary) -> void:
	var cf := ConfigFile.new()
	for k in d.keys():
		cf.set_value("display", k, d[k])
	cf.save(PATH)

static func load_or_default() -> Dictionary:
	var cf := ConfigFile.new()
	if cf.load(PATH) != OK:
		return default_dict()
	var d := default_dict()
	for k in d.keys():
		d[k] = cf.get_value("display", k, d[k])
	return d
